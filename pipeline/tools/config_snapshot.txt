@dataclass
class AppConfig:
    omp: OmpConfig = field(default_factory=OmpConfig)
    models: ModelsConfig = field(default_factory=ModelsConfig)
    concurrency: ConcurrencyConfig = field(default_factory=ConcurrencyConfig)
    research: ResearchConfig = field(default_factory=ResearchConfig)
    quality_gates: QualityGates = field(default_factory=QualityGates)
    database_path: str = "./data/merchant_intelligence.db"
    log_dir: str = "./logs"
    log_level: str = "INFO"
    export_dir: str = "./data/export"
    root: Path = field(default_factory=Path.cwd)

    def resolve(self, path: str | Path) -> Path:
        p = Path(path)
        if p.is_absolute():
            return p
        return (self.root / p).resolve()

    def safe_snapshot(self) -> dict[str, Any]:
        """Config metadata safe to persist in SQLite (no credentials)."""
        return {
            "root": str(self.root),
            "database_path": self.database_path,
            "models": {
                "discovery": self.models.discovery,
                "coordinator": self.models.coordinator,
                "analyst": self.models.analyst,
                "verifier": self.models.verifier,
                "gemini_provider": self.models.gemini_provider,
                "gpt_provider": self.models.gpt_provider,
            },
            "limits": {
                "max_parallel_agents": self.concurrency.max_parallel_agents,
                "discovery_agents": self.concurrency.discovery_agents,
                "max_luna_agents": self.concurrency.max_luna_agents,
                "max_discovery_rounds": self.research.max_discovery_rounds,
                "max_verification_rounds": self.research.max_verification_rounds,
            },
        }


def _merge(dc: Any, data: dict[str, Any]) -> Any:
    if not data:
        return dc
    kwargs = dict(dc.__dict__)
    for key, value in data.items():
        if key not in kwargs:
            continue
        current = kwargs[key]
        if hasattr(current, "__dataclass_fields__") and isinstance(value, dict):
            kwargs[key] = _merge(current, value)
        else:
            kwargs[key] = value
    return type(dc)(**kwargs)

def _apply_raw(cfg: AppConfig, raw: dict[str, Any]) -> AppConfig:
    cfg.omp = _merge(cfg.omp, raw.get("omp") or {})
    cfg.models = _merge(cfg.models, raw.get("models") or {})
    cfg.concurrency = _merge(cfg.concurrency, raw.get("concurrency") or {})
    # Accept the user's compact `agents:` naming as an alias for concurrency.
    agents = raw.get("agents") or {}
    if agents:
        mapped = {
            "discovery_count": "discovery_agents",
            "discovery_agents": "discovery_agents",
            "max_verification_agents": "max_luna_agents",
            "max_luna_agents": "max_luna_agents",
            "min_verification_agents": "min_luna_agents",
            "max_parallel": "max_parallel_agents",
        }
        values = {mapped[k]: v for k, v in agents.items() if k in mapped}
        cfg.concurrency = _merge(cfg.concurrency, values)
    cfg.research = _merge(cfg.research, raw.get("research") or {})
    cfg.quality_gates = _merge(cfg.quality_gates, raw.get("quality_gates") or {})
    # Friendly alias used in the original objective.
    model_aliases = raw.get("models") or {}
    if "discovery_coordinator" in model_aliases:
        cfg.models.coordinator = model_aliases["discovery_coordinator"]
    db = raw.get("database") or {}
    if "path" in db:
        cfg.database_path = str(db["path"])
    logging_cfg = raw.get("logging") or {}
    cfg.log_dir = str(logging_cfg.get("dir", cfg.log_dir))
    cfg.log_level = str(logging_cfg.get("level", cfg.log_level))
    export = raw.get("export") or {}
    cfg.export_dir = str(export.get("dir", cfg.export_dir))
    return cfg


def load_config(path: str | Path | None = None, *, smoke: bool = False) -> AppConfig:
    """Load explicit config or safe cwd defaults; never silently use the example."""
    cfg = AppConfig(root=Path.cwd())
    candidate = Path(path) if path else Path("config.yaml")
    if not candidate.exists():
        if path is not None:
            raise FileNotFoundError(candidate)
        if smoke:
            cfg.omp = _merge(cfg.omp, SMOKE_OVERLAY["omp"])
            cfg.concurrency = _merge(cfg.concurrency, SMOKE_OVERLAY["concurrency"])
            cfg.research = _merge(cfg.research, SMOKE_OVERLAY["research"])
            cfg.quality_gates = _merge(cfg.quality_gates, SMOKE_OVERLAY["quality_gates"])
        return cfg
    raw = yaml.safe_load(candidate.read_text(encoding="utf-8")) or {}
    if not isinstance(raw, dict):
        raise ValueError(f"configuration root must be a mapping: {candidate}")
    cfg = _apply_raw(cfg, raw)
    cfg.root = candidate.resolve().parent
    if smoke:
        cfg.omp = _merge(cfg.omp, SMOKE_OVERLAY["omp"])
        cfg.concurrency = _merge(cfg.concurrency, SMOKE_OVERLAY["concurrency"])
        cfg.research = _merge(cfg.research, SMOKE_OVERLAY["research"])
        cfg.quality_gates = _merge(cfg.quality_gates, SMOKE_OVERLAY["quality_gates"])
    return cfg
